The following examples demonstrate common practices for using one updates other orders.

Example 1: Exit Illiquid Positions in Parts

The following algorithm buys IWM each day at 10 AM Eastern Time when it has no position. After the market order fills, the algorithm places a take-profit order and a stop-loss order as OUO orders. If the take-profit order partially fills, LEAN reduces the stop-loss quantity by the same proportion, so the exit orders never sell more than the position.

public class OneUpdatesOtherExampleAlgorithm : QCAlgorithm
{
    private Symbol _iwm;

    public override void Initialize()
    {
        SetStartDate(2024, 1, 1);
        SetEndDate(2024, 3, 31);
        _iwm = AddEquity("IWM", Resolution.Minute).Symbol;
        Schedule.On(DateRules.EveryDay(_iwm), TimeRules.At(10, 0), Enter);
    }

    private void Enter()
    {
        if (Portfolio.Invested || Transactions.GetOpenOrders(_iwm).Count > 0)
        {
            return;
        }
        var fillPrice = MarketOrder(_iwm, 500).AverageFillPrice;
        OneUpdatesOtherOrder(new List<SubmitOrderRequest>
        {
            OrderFactory.LimitOrder(_iwm, -500, Math.Round(fillPrice * 1.01m, 2), tag: "Take profit"),
            OrderFactory.StopMarketOrder(_iwm, -500, Math.Round(fillPrice * 0.99m, 2), tag: "Stop loss")
        });
    }
}
class OneUpdatesOtherExampleAlgorithm(QCAlgorithm):

    def initialize(self) -> None:
        self.set_start_date(2024, 1, 1)
        self.set_end_date(2024, 3, 31)
        self._iwm = self.add_equity("IWM", Resolution.MINUTE).symbol
        self.schedule.on(self.date_rules.every_day(self._iwm), self.time_rules.at(10, 0), self._enter)

    def _enter(self) -> None:
        if self.portfolio.invested or self.transactions.get_open_orders(self._iwm):
            return
        fill_price = self.market_order(self._iwm, 500).average_fill_price
        self.one_updates_other_order([
            self.order_factory.limit_order(self._iwm, -500, round(fill_price * 1.01, 2), tag="Take profit"),
            self.order_factory.stop_market_order(self._iwm, -500, round(fill_price * 0.99, 2), tag="Stop loss")
        ])
