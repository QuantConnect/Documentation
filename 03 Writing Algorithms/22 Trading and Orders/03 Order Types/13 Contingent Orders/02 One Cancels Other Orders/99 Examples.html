The following examples demonstrate common practices for using one cancels other orders.

Example 1: Exit Each Trade at a Price Range

The following algorithm buys SPY each day at 10 AM Eastern Time when it has no position. After the market order fills, the algorithm places OCO exit orders. The take-profit order is 1% above the fill price and the stop-loss order is 1% below it. The first exit order that fills closes the position and cancels the other one.

public class OneCancelsOtherExampleAlgorithm : QCAlgorithm
{
    private Symbol _spy;

    public override void Initialize()
    {
        SetStartDate(2024, 1, 1);
        SetEndDate(2024, 3, 31);
        _spy = AddEquity("SPY", Resolution.Minute).Symbol;
        Schedule.On(DateRules.EveryDay(_spy), TimeRules.At(10, 0), Enter);
    }

    private void Enter()
    {
        if (Portfolio.Invested || Transactions.GetOpenOrders(_spy).Count > 0)
        {
            return;
        }
        var fillPrice = MarketOrder(_spy, 100).AverageFillPrice;
        OneCancelsOtherOrder(new List<SubmitOrderRequest>
        {
            OrderFactory.LimitOrder(_spy, -100, Math.Round(fillPrice * 1.01m, 2), tag: "Take profit"),
            OrderFactory.StopMarketOrder(_spy, -100, Math.Round(fillPrice * 0.99m, 2), tag: "Stop loss")
        });
    }
}
class OneCancelsOtherExampleAlgorithm(QCAlgorithm):

    def initialize(self) -> None:
        self.set_start_date(2024, 1, 1)
        self.set_end_date(2024, 3, 31)
        self._spy = self.add_equity("SPY", Resolution.MINUTE).symbol
        self.schedule.on(self.date_rules.every_day(self._spy), self.time_rules.at(10, 0), self._enter)

    def _enter(self) -> None:
        if self.portfolio.invested or self.transactions.get_open_orders(self._spy):
            return
        fill_price = self.market_order(self._spy, 100).average_fill_price
        self.one_cancels_other_order([
            self.order_factory.limit_order(self._spy, -100, round(fill_price * 1.01, 2), tag="Take profit"),
            self.order_factory.stop_market_order(self._spy, -100, round(fill_price * 0.99, 2), tag="Stop loss")
        ])

Example 2: Buy Back After the Take Profit

The following algorithm buys SPY each day at 10 AM Eastern Time when it has no position, and then places OCO exit orders. The take-profit order triggers a limit order that buys SPY back at the entry price. The buy-back order is a bracket order, so its exit orders are grandchildren of the take-profit order. If the stop loss fills first, LEAN cancels the take profit, the buy-back order, and its exit orders.

public class OneCancelsOtherGrandchildExampleAlgorithm : QCAlgorithm
{
    private Symbol _spy;

    public override void Initialize()
    {
        SetStartDate(2024, 1, 1);
        SetEndDate(2024, 3, 31);
        _spy = AddEquity("SPY", Resolution.Minute).Symbol;
        Schedule.On(DateRules.EveryDay(_spy), TimeRules.At(10, 0), Enter);
    }

    private void Enter()
    {
        if (Portfolio.Invested || Transactions.GetOpenOrders(_spy).Count > 0)
        {
            return;
        }
        var fillPrice = MarketOrder(_spy, 100).AverageFillPrice;
        var takeProfitPrice = Math.Round(fillPrice * 1.01m, 2);
        var stopLossPrice = Math.Round(fillPrice * 0.99m, 2);
        // The buy-back order and its exit orders wait for the take profit to fill.
        var buyBack = OrderFactory.LimitOrder(_spy, 100, Math.Round(fillPrice, 2)).Bracket(takeProfitPrice, stopLossPrice);
        OneCancelsOtherOrder(new List<SubmitOrderRequest>
        {
            OrderFactory.LimitOrder(_spy, -100, takeProfitPrice).Triggers(buyBack),
            OrderFactory.StopMarketOrder(_spy, -100, stopLossPrice)
        });
    }
}
class OneCancelsOtherGrandchildExampleAlgorithm(QCAlgorithm):

    def initialize(self) -> None:
        self.set_start_date(2024, 1, 1)
        self.set_end_date(2024, 3, 31)
        self._spy = self.add_equity("SPY", Resolution.MINUTE).symbol
        self.schedule.on(self.date_rules.every_day(self._spy), self.time_rules.at(10, 0), self._enter)

    def _enter(self) -> None:
        if self.portfolio.invested or self.transactions.get_open_orders(self._spy):
            return
        fill_price = self.market_order(self._spy, 100).average_fill_price
        take_profit_price = round(fill_price * 1.01, 2)
        stop_loss_price = round(fill_price * 0.99, 2)
        # The buy-back order and its exit orders wait for the take profit to fill.
        buy_back = self.order_factory.limit_order(self._spy, 100, round(fill_price, 2)).bracket(take_profit_price, stop_loss_price)
        self.one_cancels_other_order([
            self.order_factory.limit_order(self._spy, -100, take_profit_price).triggers(buy_back),
            self.order_factory.stop_market_order(self._spy, -100, stop_loss_price)
        ])
