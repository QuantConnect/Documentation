You can manually create a CrossSectionalInformationCoefficient indicator, so it doesn't automatically update. Manual indicators let you update their values with any data you choose.

Updating your indicator manually enables you to control when the indicator is updated and what data you use to update it. To manually update the indicator, call the Updateupdate method. The indicator will only be ready after you prime it with enough data.

public class CrossSectionalInformationCoefficientAlgorithm : QCAlgorithm
{
    private Symbol[] _symbols;
    private CrossSectionalInformationCoefficient _crosssectionalinformationcoefficient;

    public override void Initialize()
    {
        _symbols = new[] { "QQQ", "SPY", "IWM" }.Select(ticker => AddEquity(ticker, Resolution.Daily).Symbol).ToArray();
        _crosssectionalinformationcoefficient = new CrossSectionalInformationCoefficient(_symbols, symbol => new RateOfChange(5));
    }

    public override void OnData(Slice data)
    {
        foreach (var symbol in _symbols)
        {
            if (data.Bars.TryGetValue(symbol, out var bar))
                _crosssectionalinformationcoefficient.Update(bar);
        }

        if (_crosssectionalinformationcoefficient.IsReady)
        {
            // The current value of _crosssectionalinformationcoefficient is represented by itself (_crosssectionalinformationcoefficient)
            // or _crosssectionalinformationcoefficient.Current.Value
            Plot("CrossSectionalInformationCoefficient", "crosssectionalinformationcoefficient", _crosssectionalinformationcoefficient);
        }
    }
}
