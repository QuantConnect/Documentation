The following snippet demonstrates how to request data from the Factors dataset: 

# Factors linked to a specific stock:
self._equity = self.add_equity("AAPL", Resolution.DAILY)
self._equity.factors = self.add_data(Factors, self._equity.symbol).symbol
# Unlinked factors:
self._unlinked_factors = self.add_data(Factors, Factors.UNLINKED).symbol
# All factors:
self._universe = self.add_universe(FactorsUniverse, self._select_assets)
