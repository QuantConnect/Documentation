To add Factors data to your algorithm, call the AddDataadd_data method. For the factors linked to a specific Equity, pass the underlying Equity Symbol. For the unlinked factors, which aren't tied to any tradable security, pass Factors.UnlinkedFactors.UNLINKED. Save a reference to each dataset Symbol so you can access the data later in your algorithm.

class FactorsDataAlgorithm(QCAlgorithm):

    def initialize(self) -> None:
        self.set_start_date(2014, 3, 25)
        self.set_end_date(2014, 4, 8)
        self.set_cash(100000)
        self._equity = self.add_equity("SPY", Resolution.DAILY)
        self._equity.factors = self.add_data(Factors, self._equity.symbol).symbol
        self._unlinked_factors = self.add_data(Factors, Factors.UNLINKED).symbol
public class FactorsDataAlgorithm : QCAlgorithm
{
    private Equity _equity;
    private Symbol _equityFactors, _unlinkedFactors;

    public override void Initialize()
    {
        SetStartDate(2014, 3, 25);
        SetEndDate(2014, 4, 8);
        SetCash(100000);
        _equity = AddEquity("SPY", Resolution.Daily);
        _equityFactors = AddData<Factors>(_equity.Symbol).Symbol;
        _unlinkedFactors = AddData<Factors>(Factors.Unlinked).Symbol;
    }
}
