To select a dynamic universe of US Equities based on factor values, call the AddUniverseadd_universe method with the FactorsUniverse class and a selection function.

def initialize(self) -> None:
    self._universe = self.add_universe(FactorsUniverse, self._select_assets)

def _select_assets(self, data: List[FactorsUniverse]) -> List[Symbol]:
    ust_yield = None
    positive = []
    negative = []
    for factors in data:
        if Factors.UST_YIELD_1_MONTH in factors.keys:
            ust_yield = factors[Factors.UST_YIELD_1_MONTH]
        elif Factors.ROC_1_DAY in factors.keys:
            if factors[Factors.ROC_1_DAY] > 0:
                positive.append(factors.symbol)
            else:
                negative.append(factors.symbol)
    return positive if ust_yield < 4 else negative
private Universe _universe;

public override void Initialize()
{
    _universe = AddUniverse<FactorsUniverse>(SelectAssets);
}

private IEnumerable<Symbol> SelectAssets(IEnumerable<BaseData> data)
{
    decimal? ustYield = null;
    var positive = new List<Symbol>();
    var negative = new List<Symbol>();
    foreach (var factors in data.OfType<FactorsUniverse>())
    {
        if (factors.ContainsKey(Factors.UstYield1Month))
        {
            ustYield = factors[Factors.UstYield1Month];
        }
        else if (factors.ContainsKey(Factors.Roc1Day))
        {
            if (factors[Factors.Roc1Day] > 0)
            {
                positive.Add(factors.Symbol);
            }
            else
            {
                negative.Add(factors.Symbol);
            }
        }
    }
    return ustYield < 4 ? positive : negative;
}
