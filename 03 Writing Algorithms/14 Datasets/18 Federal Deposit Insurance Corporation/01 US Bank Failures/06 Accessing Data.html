To get the current US Bank Failures data, index the current Slice with the dataset Symbol. Slice objects deliver unique events to your algorithm as they happen, but the Slice may not contain data for your dataset at every time step. To avoid issues, check if the Slice contains the data you want before you index it.

def on_data(self, slice: Slice) -> None:
    failures = slice.get(FDICBankFailures).get(self._dataset_symbol)
    if failures:
        for failure in failures.data:
            self.log(f"{failure.institution_name} ({failure.state}): assets {failure.total_assets}")
public override void OnData(Slice slice)
{
    if (slice.TryGet<FDICBankFailures>(_datasetSymbol, out var failures))
    {
        foreach (FDICBankFailure failure in failures.Data)
        {
            Log($"{failure.InstitutionName} ({failure.State}): assets {failure.TotalAssets}");
        }
    }
}
