To select a dynamic universe of US Equities based on US Bank Financials data, call the AddUniverseadd_universe method with the FDICBankFinancialsUniverse class and a selection function.

def initialize(self) -> None:
    self._universe = self.add_universe(FDICBankFinancialsUniverse, self.universe_selection)

def universe_selection(self, alt_coarse: List[FDICBankFinancialsUniverse]) -> List[Symbol]:
    return list({d.symbol for d in alt_coarse if d.domestic_deposits and (d.estimated_uninsured_deposits or 0) / d.domestic_deposits < 0.3})
private Universe _universe;

public override void Initialize()
{
    _universe = AddUniverse<FDICBankFinancialsUniverse>(altCoarse =>
        (from d in altCoarse.OfType<FDICBankFinancialsUniverse>()
        where d.DomesticDeposits > 0m && (d.EstimatedUninsuredDeposits ?? 0m) / d.DomesticDeposits < 0.3m
        select d.Symbol).Distinct());
}
