To get the current US Bank Financials data, index the current Slice with the dataset Symbol. Slice objects deliver unique events to your algorithm as they happen, but the Slice may not contain data for your dataset at every time step. To avoid issues, check if the Slice contains the data you want before you index it.

def on_data(self, slice: Slice) -> None:
    reports = slice.get(FDICBankFinancials).get(self._dataset_symbol)
    if reports:
        for report in reports.data:
            self.log(f"{report.institution_name} {report.report_date}: {report.estimated_uninsured_deposits}")
public override void OnData(Slice slice)
{
    if (slice.TryGet<FDICBankFinancials>(_datasetSymbol, out var reports))
    {
        foreach (FDICBankFinancial report in reports.Data)
        {
            Log($"{report.InstitutionName} {report.ReportDate}: {report.EstimatedUninsuredDeposits}");
        }
    }
}

To iterate through all of the dataset objects in the current Slice, call the Getget method.

def on_data(self, slice: Slice) -> None:
    for dataset_symbol, reports in slice.get(FDICBankFinancials).items():
        for report in reports.data:
            self.log(f"{dataset_symbol} {report.institution_name}: {report.total_deposits}")
public override void OnData(Slice slice)
{
    foreach (var kvp in slice.Get<FDICBankFinancials>())
    {
        var datasetSymbol = kvp.Key;
        foreach (FDICBankFinancial report in kvp.Value.Data)
        {
            Log($"{datasetSymbol} {report.InstitutionName}: {report.TotalDeposits}");
        }
    }
}
