To get the current USPTO Patent Maintenance data, index the current Slice with the dataset Symbol. Slice objects deliver unique events to your algorithm as they happen, but the Slice may not contain data for your dataset at every time step. To avoid issues, check if the Slice contains the data you want before you index it.

def on_data(self, slice: Slice) -> None:
    if slice.contains_key(self._patents):
        data_point = slice[self._patents]
        self.log(f"{self._patents} expired patents at {slice.time}: {data_point.patents_expired}")
public override void OnData(Slice slice)
{
    if (slice.ContainsKey(_patents))
    {
        var dataPoint = slice[_patents];
        Log($"{_patents} expired patents at {slice.Time}: {dataPoint.PatentsExpired}");
    }
}

To iterate through all of the dataset objects in the current Slice, call the Getget method.

def on_data(self, slice: Slice) -> None:
    for dataset_symbol, data_point in slice.get(USPTOPatentMaintenance).items():
        self.log(f"{dataset_symbol} expired patents at {slice.time}: {data_point.patents_expired}")
public override void OnData(Slice slice)
{
    foreach (var kvp in slice.Get<USPTOPatentMaintenance>())
    {
        var datasetSymbol = kvp.Key;
        var dataPoint = kvp.Value;
        Log($"{datasetSymbol} expired patents at {slice.Time}: {dataPoint.PatentsExpired}");
    }
}
